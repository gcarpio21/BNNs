\newcommand{\defZ}[3][]{\defSet[#1]{Z}{#2,#3}}
\begin{definition}[Zonotope~{\cite[Def. 1]{girard2005reachability}}]
    \label{def:zonotope}
    Given a center vector $c\in\mathbb{R}^n$ and a generator matrix $G\in\mathbb{R}^{n\times q}$,
    a zonotope is defined as
    \begin{equation*}
        \mathcal{Z} = \defZ{c}{G} \coloneqq \left\{ c + \sum_{j=1}^{q} \beta_j G_{(\cdot,
        j)}\ \middle|\ \beta_j \in \left[-1,1\right] \right\}.
    \end{equation*}
\end{definition}
