\begin{frame}{Background: Bayesian Deep Learning}
    \begin{itemize}
        \item \textbf{Goal:} Replace point estimates $\bm{\theta}$ with a posterior $p(\bm{\theta} \mid \mathcal{D})$.
        \item \textbf{Laplace Approximation:} Post-hoc Gaussian fit around the MAP estimate using the Hessian (GGN).
        \item \textbf{Variational Inference:} Optimization-based approach maximizing the ELBO using the reparameterization trick.
        \item \textbf{Metrics of interest:} Brier Score, ECE, Uncertainty Calibration (Expected vs Actual), and OOD robustness.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Implementation: Libraries}
\begin{block}{laplace-torch (Post-hoc)}
\begin{lstlisting}[language=Python, basicstyle=\ttfamily\tiny]
la = Laplace(model, 'classification', 
             subset_of_weights='last_layer', 
             hessian_structure='kron')
la.fit(train_loader)
la.optimize_prior_precision(method='gridsearch') # or 'marglik'
\end{lstlisting}
\end{block}

\begin{block}{bayesian-torch (Variational)}
\begin{lstlisting}[language=Python, basicstyle=\ttfamily\tiny]
dnn_to_bnn(net, {"type": "Reparameterization", # or "Flipout"
                 "moped_enable": True,
                 "moped_delta": 0.5})
# Loss: ELBO or ELBO + AvUC (Accuracy vs Uncertainty)
\end{lstlisting}
\end{block}
\end{frame}
